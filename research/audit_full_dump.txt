 160:     def resolve(self, path: str) -> Path:
 161:         """Strict path resolution: resolves path against cwd without fuzzy matching."""
 162:         p = Path(path).expanduser()
 163:         if not p.is_absolute():
 164:             p = self.cwd / p
 165:         return p.resolve()

 167:     def resolve_resilient(self, path: str) -> tuple[Path, str | None]:
 168:         """Resolve a path with safe relative auto-correction:
 169:         1. If exact path exists, return it immediately.
 170:         2. Absolute paths are NEVER fuzzy-resolved (avoids cross-project collisions).
 171:         3. Never fuzzy-resolve if cwd is root ('/') or the user's root home ('~').
 172:         4. If path is a relative basename (e.g. 'tui.py') and exactly ONE match
 173:            exists within project cwd, auto-resolve it."""
 174:         raw_p = Path(path).expanduser()
 175:         is_abs = raw_p.is_absolute()
 176:         resolved = (raw_p if is_abs else (self.cwd / raw_p)).resolve()
 177:         if resolved.exists():
 178:             return resolved, None
 179: 
 180:         # Absolute paths must NEVER be fuzzy-redirected to existing files elsewhere
 181:         if is_abs:
 182:             return resolved, None
 183: 
 184:         # Disallow global scans from home root or filesystem root
 185:         if self.cwd in (Path.home(), Path("/"), Path("/home")):
 186:             return resolved, None
 187: 
 188:         # Attempt unique fuzzy resolution within self.cwd for relative basenames
 189:         target_name = raw_p.name
 190:         if target_name and len(raw_p.parts) == 1 and not path.startswith(".."):
 191:             matches = []
 192:             try:
 193:                 for candidate in self.cwd.rglob(target_name):
 194:                     parts = candidate.parts
 195:                     if any(part.startswith(".") or part in ("__pycache__", "venv", ".venv", "node_modules") for part in parts):
 196:                         continue
 197:                     if candidate.is_file():
 198:                         matches.append(candidate)
 199:                         if len(matches) > 3:
 200:                             break
 201:             except Exception:
 202:                 pass
 203:             if len(matches) == 1:
 204:                 match = matches[0]
 205:                 rel = match.relative_to(self.cwd)
 206:                 return match, f"[auto-resolved '{path}' -> '{rel}']"
 207: 
 208:         return resolved, None

 234: def tool_read(fs: FS, path: str, offset: int = 1, limit: int = 200,
 235:               full: bool = False) -> tuple[str, dict]:
 236:     """Read a file or inspect multimodal media (images/audio/binary).
 237:     Supports auto-path resolution, numbered text slices, and native multimodal
 238:     payload attachment for vision/audio models on VSLLM."""
 239:     p, note = fs.resolve_resilient(path)
 240:     res_prefix = (note + "\n") if note else ""
 241: 
 242:     if p.is_dir():
 243:         entries = sorted(os.listdir(p))[:200]
 244:         return f"{res_prefix}{p}/ (directory)\n" + "\n".join(entries), {}
 245:     if not p.exists():
 246:         near = difflib.get_close_matches(str(p), [str(x) for x in p.parent.glob("*")], n=3) if p.parent.exists() else []
 247:         return (f"error: no such file: {p}"
 248:                 + (f"\ndid you mean: {', '.join(near)}" if near else "")), {}
 249: 
 250:     size = p.stat().st_size
 251:     suffix = p.suffix.lower()
 252: 
 253:     # Multimodal Media: Images
 254:     IMAGE_EXTS = {".png", ".jpg", ".jpeg", ".webp", ".gif"}
 255:     if suffix in IMAGE_EXTS:
 256:         import base64, mimetypes
 257:         mime = mimetypes.guess_type(str(p))[0] or f"image/{suffix.lstrip('.')}"
 258:         if size <= 10 * 1024 * 1024:  # up to 10MB
 259:             try:
 260:                 b64_data = base64.b64encode(p.read_bytes()).decode("ascii")
 261:                 meta = {"media": {"type": "image", "mime": mime, "data": b64_data, "path": str(p)}}
 262:                 return f"{res_prefix}[image: {p.name} ({mime}, {size:,} bytes) — visual content attached for multimodal models]", meta
 263:             except Exception as e:
 264:                 pass
 265:         return f"{res_prefix}[image file: {p.name} ({mime}, {size:,} bytes) — too large for in-line attachment]", {}
 266: 
 267:     # Multimodal Media: Audio
 268:     AUDIO_EXTS = {".mp3", ".wav", ".ogg", ".m4a", ".flac"}
 269:     if suffix in AUDIO_EXTS:
 270:         import mimetypes
 271:         mime = mimetypes.guess_type(str(p))[0] or f"audio/{suffix.lstrip('.')}"
 272:         return f"{res_prefix}[audio file: {p.name} ({mime}, {size:,} bytes) — this adapter does not transcribe audio; use a configured transcription tool]", {}
 273: 
 274:     # Pure binary files (.so, .bin, .pyc, .exe, zip/tar, or null-byte detection)
 275:     BINARY_EXTS = {".so", ".dylib", ".dll", ".bin", ".exe", ".pyc", ".tar", ".gz", ".zip", ".7z", ".pdf"}
 276:     is_binary = suffix in BINARY_EXTS
 277:     if not is_binary and size > 0:
 278:         try:
 279:             with open(p, "rb") as bf:
 280:                 chunk = bf.read(1024)
 281:                 if _is_binary_bytes(chunk):
 282:                     is_binary = True
 283:         except Exception:
 284:             pass
 285: 
 286:     if is_binary:
 287:         import mimetypes
 288:         mime = mimetypes.guess_type(str(p))[0] or "application/octet-stream"
 289:         return (f"{res_prefix}[binary file: {p.name} ({mime}, {size:,} bytes) — "
 290:                 f"raw binary read skipped to prevent context window corruption]"), {}
 291: 
 292:     # Standard Text File
 293:     if full:
 294:         text = p.read_text(encoding="utf-8", errors="strict")
 295:         if len(text.splitlines()) <= 2000:
 296:             return f"{res_prefix}{text}", {}
 297:         return (f"{res_prefix}error: file too large for full read ({len(text.splitlines())} lines). "
 298:                 f"Use offset/limit instead."), {}
 299:     return f"{res_prefix}{_numbered(p, offset, limit)}", {}

 301: def tool_write(fs: FS, session, path: str, content: str) -> tuple[str, dict]:
 302:     p = fs.resolve(path)
 303:     # Pre-flight: refuse a .py write that wouldn't compile BEFORE touching disk (F-D).
 304:     # A broken file left on disk is far worse than a refused write.
 305:     if p.suffix == ".py":
 306:         ok, err = _py_compiles_str(content)
 307:         if not ok:
 308:             return (f"error: refusing to write {p} — content does not compile: {err}. "
 309:                     f"No changes made. Fix the syntax and try again."), {"path": str(p)}
 310:     p.parent.mkdir(parents=True, exist_ok=True)
 311:     res = _locked_update(p, lambda _src: content,
 312:                          before_write=lambda: session.checkpoint([str(p)], cwd=str(fs.cwd)))
 313:     old, _new = res if res else ("", content)
 314:     diff = _unified_diff(p, old, content)
 315:     msg = f"wrote {p} ({len(content)} bytes)"
 316:     return msg, {"diff": diff, "path": str(p)}

 319: def _locked_update(p: Path, fn, before_write=None) -> tuple[str, str] | None:
 320:     """Whole read-modify-write cycle under an exclusive sidecar lock.
 321:     fn(src) -> new_src, or None to refuse (file untouched). Returns
 322:     (old_src, new_src) on success. The atomic rename makes a crash
 323:     mid-write leave the target intact. Limit: only writers that also take
 324:     this lock (kern tools) are serialized — a non-cooperating external
 325:     process can still race; the edit precondition catches the common case
 326:     by refusing on drifted content."""
 327:     lock = KERN_HOME / "locks" / (path_key(p) + ".lock")
 328:     with file_lock(lock):
 329:         existed = p.exists()
 330:         raw = p.read_bytes() if existed else b""
 331:         src = raw.decode("utf-8")
 332:         newline = "\r\n" if b"\r\n" in raw else "\n"
 333:         src = src.replace("\r\n", "\n")
 334:         new = fn(src)
 335:         if new is None:
 336:             return None
 337:         if new != src or not existed:
 338:             if before_write:
 339:                 before_write()
 340:             atomic_write(p, new.replace("\r\n", "\n").replace("\n", newline))
 341:         return src, new

 354: def _relocate_line_range(flines: list[str], expected: str, start_line: int, end_line: int) -> tuple[int, int, str] | None:
 355:     """Find relocated position of expected lines if file has drifted due to prior edits."""
 356:     exp_lines = expected.splitlines()
 357:     exp_len = len(exp_lines)
 358:     if exp_len == 0 or len(flines) < exp_len:
 359:         return None
 360: 
 361:     # 1. Exact match search across all windows of length exp_len
 362:     exact_matches: list[int] = []
 363:     for i in range(len(flines) - exp_len + 1):
 364:         if flines[i : i + exp_len] == exp_lines:
 365:             exact_matches.append(i + 1)  # 1-based start line
 366: 
 367:     if len(exact_matches) == 1:
 368:         s = exact_matches[0]
 369:         return (s, s + exp_len - 1, f"exact match (drifted {s - start_line:+d} lines)")
 370:     elif len(exact_matches) > 1:
 371:         exact_matches.sort(key=lambda x: abs(x - start_line))
 372:         d0 = abs(exact_matches[0] - start_line)
 373:         d1 = abs(exact_matches[1] - start_line)
 374:         if d0 < d1 and d0 <= 200:
 375:             s = exact_matches[0]
 376:             return (s, s + exp_len - 1, f"closest exact match (drifted {s - start_line:+d} lines)")
 377: 
 378:     # 2. Whitespace-tolerant match (trailing whitespace / CRLF differences)
 379:     norm_exp = [l.rstrip() for l in exp_lines]
 380:     ws_matches: list[int] = []
 381:     for i in range(len(flines) - exp_len + 1):
 382:         if [l.rstrip() for l in flines[i : i + exp_len]] == norm_exp:
 383:             ws_matches.append(i + 1)
 384: 
 385:     if len(ws_matches) == 1:
 386:         s = ws_matches[0]
 387:         return (s, s + exp_len - 1, f"whitespace-tolerant match (drifted {s - start_line:+d} lines)")
 388:     elif len(ws_matches) > 1:
 389:         ws_matches.sort(key=lambda x: abs(x - start_line))
 390:         d0 = abs(ws_matches[0] - start_line)
 391:         d1 = abs(ws_matches[1] - start_line)
 392:         if d0 < d1 and d0 <= 200:
 393:             s = ws_matches[0]
 394:             return (s, s + exp_len - 1, f"closest whitespace match (drifted {s - start_line:+d} lines)")
 395: 
 396:     return None

 399: def tool_edit(fs: FS, session, path: str, old_str: str = "", new_str: str = "",
 400:               start_line: int = 0, end_line: int = 0,
 401:               expected: str = "", occurrence: int = 0) -> tuple[str, dict]:
 402:     """Edit a file. By default replaces exact old_str with new_str.
 403:     If old_str fails and start_line/end_line are given, replaces that line
 404:     range with new_str instead (1-based, inclusive).
 405: 
 406:     Robustness (fixes observed live this session):
 407:       - old_str/new_str default to "" so line-range mode never TypeErrors (F-A).
 408:       - When exact old_str matches nothing, a whitespace-tolerant fallback maps the
 409:         match back to source, fixing pure indentation/whitespace drift (F-B).
 410:       - occurrence selects which match to replace when old_str is non-unique (F-C);
 411:         occurrence=0 with multiple matches refuses and reports the count.
 412:       - .py edits are compile-checked BEFORE write and rolled back if they break
 413:         syntax (F-D) — a broken file is never left on disk."""
 414:     p = fs.resolve(path)
 415:     if not p.exists():
 416:         return f"error: no such file: {p}. Use write() to create it.", {}

 418:     lines = src.splitlines()
 419: 
 420:     # Line-range mode: use when old_str is empty or fails.
 421:     # Replaces lines by position, with automatic line drift relocation if prior
 422:     # edits moved the lines, plus rich contextual feedback on failure.
 423:     if start_line > 0 and end_line > 0:
 424:         if not expected:

 727: def tool_fetch(url: str, max_chars: int = 12000, cache: dict | None = None) -> tuple[str, dict]:
 728:     import httpx
 729:     if not url.startswith(("http://", "https://")):
 730:         url = "https://" + url
 731:     # Session cache: a 30-step research turn often re-fetches the same URL
 732:     # after compaction evicts the earlier result. The cache lives in the
 733:     # session's scratch dir (caller wires it) — zero LLM cost, fewer
 734:     # tool->model round trips.
 735:     ckey = None
 736:     if cache is not None:
 737:         ckey = f"{url}::{max_chars}"
 738:         if ckey in cache:
 739:             return cache[ckey], {}

 741:         r = httpx.get(url, timeout=20, follow_redirects=True,

 162: @dataclass
 163: class Doc:
 164:     id: str
 165:     text: str
 166:     tokens: list[str]
 167:     pinned: bool = False
 168:     ts: float = 0.0
 169:     source: str = ""

 172: class BM25Index:
 173:     """In-memory BM25 over a small corpus (ledger entries + atoms). Rebuilt cheaply;
 174:     for an agent session the corpus is small enough that this is faster than any
 175:     embedding round-trip — and it is exact/deterministic, not approximate."""
 176: 
 177:     def __init__(self, k1: float = 1.5, b: float = 0.75):
 178:         self.k1, self.b = k1, b
 179:         self.docs: list[Doc] = []
 180:         self.df: Counter = Counter()
 181:         self.avgdl = 0.0
 182: 
 183:     def add(self, doc: Doc) -> None:
 184:         doc.tokens = doc.tokens or tokenize(doc.text)
 185:         self.docs.append(doc)
 186:         for t in set(doc.tokens):
 187:             self.df[t] += 1
 188:         self.avgdl = sum(len(d.tokens) for d in self.docs) / max(1, len(self.docs))
 189: 
 190:     def build(self, docs) -> None:
 191:         self.docs, self.df = [], Counter()
 192:         for d in docs:
 193:             self.add(d)
 194: 
 195:     def _idf(self, term: str) -> float:
 196:         n, df = len(self.docs), self.df.get(term, 0)
 197:         return math.log(1 + (n - df + 0.5) / (df + 0.5))
 198: 
 199:     def score(self, query: str, doc: Doc) -> float:
 200:         qtokens = tokenize(query)
 201:         if not qtokens or not doc.tokens:
 202:             return 0.0
 203:         tf = Counter(doc.tokens)
 204:         dl = len(doc.tokens)
 205:         s = 0.0
 206:         for t in qtokens:
 207:             if t not in tf:
 208:                 continue
 209:             f = tf[t]
 210:             idf = self._idf(t)
 211:             s += idf * (f * (self.k1 + 1)) / (f + self.k1 * (1 - self.b + self.b * dl / max(self.avgdl, 1e-9)))
 212:         return s
 213: 
 214:     def search(self, query: str, *, now: float | None = None, half_life_s: float = 86400.0,
 215:                recency_weight: float = 0.15, pin_boost: float = 2.0) -> list[tuple[Doc, float]]:
 216:         """Rank docs by BM25 + recency-decay (rank only — never deletes) + pin boost.
 217:         Salience = BM25 relevance * recency factor, pins float to the top (P-design L3)."""
 218:         now = now if now is not None else time.time()
 219:         out = []
 220:         for d in self.docs:
 221:             base = self.score(query, d)
 222:             if base <= 0:
 223:                 continue                      # no term overlap = irrelevant; never surface
 224:             age = max(0.0, now - (d.ts or now))
 225:             recency = 0.5 ** (age / half_life_s) if d.ts else 1.0
 226:             score = base * (1 - recency_weight) + base * recency_weight * recency
 227:             if d.pinned:
 228:                 score *= pin_boost
 229:             out.append((d, score))
 230:         out.sort(key=lambda x: x[1], reverse=True)
 231:         return out

 238: def filter_against_context(candidates: list[tuple[Doc, float]],
 239:                            context_text: str,
 240:                            *, max_items: int = 6, max_per_source: int = 2,
 241:                            token_budget: int = 800) -> list[Doc]:
 242:     """Before retrieved memory re-enters context: drop items whose content is already
 243:     present in context (near-dup on normalized tokens) and cap per-source + total, so
 244:     the agent's own recycled notes cannot feed back and cause loops (O1)."""
 245:     ctx_keys = set()
 246:     ctx_norm = []
 247:     for line in context_text.splitlines():
 248:         k = _norm_key(line)
 249:         if k:
 250:             ctx_keys.add(k)
 251:             ctx_norm.append(k)
 252:     ctx_blob = " ".join(ctx_norm)
 253:     # O1 containment threshold (env-tunable). A candidate is an "echo" only when
 254:     # most of its *distinctive* content is already visible in the context.
 255:     try:
 256:         o1_threshold = float(os.environ.get('KERN_O1_THRESHOLD', '0.6'))
 257:     except ValueError:
 258:         o1_threshold = 0.6
 259:     # IDF over the candidate set + context lines: rare tokens weigh more, so a doc
 260:     # sharing only common words with the context is NOT treated as an echo, while a
 261:     # doc whose distinctive tokens are all already visible IS dropped.
 262:     df: Counter = Counter()
 263:     for line in ctx_norm:
 264:         for t in set(line.split()):
 265:             df[t] += 1
 266:     cand_tok_sets = []
 267:     for doc, _s in candidates:
 268:         ts = set(doc.tokens or tokenize(doc.text))
 269:         cand_tok_sets.append(ts)
 270:         for t in ts:
 271:             df[t] += 1
 272:     n_docs = max(1, len(ctx_norm) + len(cand_tok_sets))
 273:     idf = {t: math.log(1 + n_docs / df[t]) for t in df}
 274:     seen: set[str] = set()
 275:     per_source: Counter = Counter()
 276:     chosen: list[Doc] = []
 277:     budget = token_budget
 278:     for (doc, _score), tok_set in zip(candidates, cand_tok_sets):
 279:         k = _norm_key(doc.text)
 280:         toks = doc.tokens or tokenize(doc.text)
 281:         if not k:
 282:             continue
 283:         # O1: drop if (a) exact/substring match, OR (b) a highly distinctive token
 284:         # (a rare compound identifier like a codename, version, or path — the thing
 285:         # that makes the fact worth recalling) is already visible in the context.
 286:         if k in ctx_keys or (len(k) >= 12 and k in ctx_blob):
 287:             continue                          # already in context -> circular feedback
 288:         if ctx_blob:
 289:             # distinctive = long, or compound (hyphenated/slashed/dotted) tokens.
 290:             distinctive = [t for t in set(toks)
 291:                            if len(t) >= 8 or ('-' in t or '/' in t or '.' in t)]
 292:             # Echo only when the distinctive identifier is visible AND most of the
 293:             # doc's content tokens are too — so a NEW fact that merely references a
 294:             # known artifact survives, while a re-statement of it is dropped.
 295:             if distinctive and all(t in ctx_blob for t in distinctive):
 296:                 present = sum(1 for t in set(toks) if t in ctx_blob)
 297:                 if present / max(1, len(set(toks))) >= 0.5:
 298:                     continue                      # its identifying token(s) are visible
 299:         # IDF-weighted containment: also drop when the visible context already covers
 300:         # most of this doc's distinctive content weight (env-tunable).
 301:         if tok_set and ctx_blob:
 302:             total_w = sum(idf.get(t, 0.0) for t in tok_set)
 303:             present_w = sum(idf.get(t, 0.0) for t in tok_set if t in ctx_blob)
 304:             if total_w > 0 and present_w / total_w >= o1_threshold:
 305:                 continue
 306:         if k in seen:
 307:             continue                          # duplicate within this retrieval batch
 308:         if per_source[doc.source] >= max_per_source:
 309:             continue                          # one source may not dominate
 310:         cost = len(toks) + 8
 311:         if cost > budget:
 312:             continue                          # token budget guard
 313:         seen.add(k)
 314:         per_source[doc.source] += 1
 315:         budget -= cost
 316:         chosen.append(doc)
 317:         if len(chosen) >= max_items:
 318:             break
 319:     return chosen

 326: def recall(query: str, *, atoms: list[dict] | None = None,
 327:            ledger: list[LedgerEntry] | None = None,
 328:            context_text: str = "", max_items: int = 6,
 329:            token_budget: int = 800, now: float | None = None) -> list[dict]:
 330:     """The hot-path recall. Deterministic BM25 over atoms + structured ledger, then
 331:     anti-circularity filtering against the current context. Returns ranked, deduped,
 332:     budget-capped memory items ready to inject. ZERO model requests.
 333: 
 334:     atoms: [{'id','text','pinned','ts','source'}...]  (long-term facts)
 335:     ledger: [LedgerEntry...]  (structured episode records)
 336:     """
 337:     docs: list[Doc] = []
 338:     for a in (atoms or []):
 339:         docs.append(Doc(id=str(a.get("id", "")), text=str(a.get("text", "")),
 340:                         tokens=[], pinned=bool(a.get("pinned")),
 341:                         ts=float(a.get("ts", 0.0)), source=str(a.get("source", "atom"))))
 342:     for ent in (ledger or []):
 343:         blob_parts = ent.decisions + ent.open_threads + ent.artifacts
 344:         if not blob_parts:
 345:             continue
 346:         docs.append(Doc(id=f"ledger:{ent.n}", text=" | ".join(blob_parts) or ent.text,
 347:                         tokens=[], pinned=False, ts=0.0, source=f"event:{ent.n}"))
 348:     if not docs:
 349:         return []
 350:     idx = BM25Index()
 351:     idx.build(docs)
 352:     ranked = idx.search(query, now=now)
 353:     chosen = filter_against_context(ranked, context_text, max_items=max_items,
 354:                                     token_budget=token_budget)
 355:     return [{"id": d.id, "text": d.text, "source": d.source, "pinned": d.pinned}
 356:             for d in chosen]
